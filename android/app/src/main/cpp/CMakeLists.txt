cmake_minimum_required(VERSION 3.22)
project(kai_mobile LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    # Keep inference optimized even in the sideloadable Java debug preview.
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
add_subdirectory(../../../../vendor/llama.cpp ${CMAKE_BINARY_DIR}/llama)
if(ANDROID)
    add_library(kai-inference SHARED engine.cpp bridge.cpp)
    target_link_libraries(kai-inference PRIVATE llama android log)
    target_link_options(kai-inference PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(kai-engine-smoke engine.cpp smoke.cpp)
    target_link_libraries(kai-engine-smoke PRIVATE llama)
endif()
